# EXP-43: Car Testing Data - Standard Deviation & Variance Analysis

cars_data <- data.frame(
  CarID = 1:9,
  AvgSpeed = c(78, 65, 82, 90, 70, 85, 60, 75, 88),
  TotalTime = c(2.5, 3.2, 2.1, 1.8, 3.0, 2.2, 3.8, 2.7, 2.0)
)

cat("=====================================================\n")
cat("  EXP-43: CAR TESTING - VARIANCE & STANDARD DEVIATION\n")
cat("=====================================================\n\n")

cat("Car Testing Dataset (9 randomly selected cars):\n")
print(cars_data)
cat("\n")

sd_speed <- sd(cars_data$AvgSpeed)
var_speed <- var(cars_data$AvgSpeed)

sd_time <- sd(cars_data$TotalTime)
var_time <- var(cars_data$TotalTime)

cat("-----------------------------------------------------\n")
cat("STATISTICAL RESULTS:\n")
cat("-----------------------------------------------------\n")
cat(sprintf("1. AvgSpeed Standard Deviation: %.4f km/h\n", sd_speed))
cat(sprintf("2. AvgSpeed Variance:           %.4f\n\n", var_speed))

cat(sprintf("3. TotalTime Standard Deviation: %.4f hours\n", sd_time))
cat(sprintf("4. TotalTime Variance:           %.4f\n", var_time))
