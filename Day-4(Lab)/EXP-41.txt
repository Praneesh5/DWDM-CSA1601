# EXP-41: ARFF Dataset, Apriori vs FP-Growth Unique Rules Comparison

transactions <- list(
  T1 = c("Item1", "Item2", "Item5"),
  T2 = c("Item2", "Item4"),
  T3 = c("Item2", "Item3"),
  T4 = c("Item1", "Item2", "Item4"),
  T5 = c("Item1", "Item3"),
  T6 = c("Item2", "Item3")
)

N <- length(transactions)
min_sup <- 2        # Min_sup = 2
min_conf <- 0.50    # Confidence = 50%

cat("=====================================================\n")
cat("  EXP-41: APRIORI vs FP-GROWTH UNIQUE RULES MINING\n")
cat("=====================================================\n\n")

cat("--- ARFF DATASET FORMAT ---\n")
all_items <- c("Item1", "Item2", "Item3", "Item4", "Item5")
arff <- "@relation market_items\n\n"
for (it in all_items) arff <- paste0(arff, "@attribute ", it, " {t, f}\n")
arff <- paste0(arff, "\n@data\n")
for (id in names(transactions)) {
  t <- transactions[[id]]
  vals <- sapply(all_items, function(x) if (x %in% t) "t" else "f")
  arff <- paste0(arff, paste(vals, collapse=","), "\n")
}
cat(arff, "\n")

freq1 <- table(unlist(transactions))
L1 <- freq1[freq1 >= min_sup]
cat("Frequent 1-Itemsets (L1):\n")
print(L1)

get_support <- function(itemset, trans_list) {
  sum(sapply(trans_list, function(t) all(itemset %in% t)))
}

item_names <- names(L1)
pairs <- combn(item_names, 2, simplify = FALSE)
C2_sup <- sapply(pairs, get_support, trans_list = transactions)
names(C2_sup) <- sapply(pairs, function(p) paste(p, collapse = ", "))
L2_sup <- C2_sup[C2_sup >= min_sup]

cat("\nFrequent 2-Itemsets (L2):\n")
print(L2_sup)

cat("\n--- APRIORI & FP-GROWTH GENERATED RULES ---\n")
rules <- character()
for (name in names(L2_sup)) {
  items_in_set <- unlist(strsplit(name, ", "))
  sup_AB <- L2_sup[name]
  for (i in 1:length(items_in_set)) {
    lhs <- items_in_set[i]
    rhs <- setdiff(items_in_set, lhs)
    sup_A <- get_support(lhs, transactions)
    conf <- sup_AB / sup_A
    if (conf >= min_conf) {
      r_str <- sprintf("{%s} => {%s} [Supp: %d/6, Conf: %.0f%%]", lhs, rhs, sup_AB, conf*100)
      rules <- c(rules, r_str)
    }
  }
}

for (r in rules) cat(r, "\n")
cat(sprintf("\nTotal Unique Rules Discovered: %d\n", length(unique(rules))))
