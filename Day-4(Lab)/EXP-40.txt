# EXP-40: Decision Tree to ARFF, Rule Induction & Confusion Matrix

# Decision Tree from Image 3:
# Root: Height > 180cm -> Yes: Male
# No -> Weight > 80kg -> Yes: Male, No: Female

df <- data.frame(
  Height = c(185, 175, 170, 182, 165, 178, 190, 160, 172, 188),
  Weight = c(75, 85, 60, 90, 55, 82, 70, 50, 65, 95),
  Gender = c("Male", "Male", "Female", "Male", "Female", "Male", "Male", "Female", "Female", "Male")
)

cat("=====================================================\n")
cat("  EXP-40: ARFF CREATION, RULE INDUCTION & EVALUATION\n")
cat("=====================================================\n\n")

cat("--- a) ARFF DATASET FORMAT ---\n")
arff <- "@relation person_gender_tree\n\n@attribute Height numeric\n@attribute Weight numeric\n@attribute Gender {Male, Female}\n\n@data\n"
for (i in 1:nrow(df)) {
  arff <- paste0(arff, df$Height[i], ",", df$Weight[i], ",", df$Gender[i], "\n")
}
cat(arff, "\n")

cat("--- b) RULE-BASED INDUCTION RULES ---\n")
cat("Rule 1: IF Height > 180 THEN Gender = Male\n")
cat("Rule 2: IF Height <= 180 AND Weight > 80 THEN Gender = Male\n")
cat("Rule 3: IF Height <= 180 AND Weight <= 80 THEN Gender = Female\n\n")

predict_gender <- function(h, w) {
  if (h > 180) return("Male")
  else if (w > 80) return("Male")
  else return("Female")
}

df$Predicted <- sapply(1:nrow(df), function(i) predict_gender(df$Height[i], df$Weight[i]))

cat("--- c) CONFUSION MATRIX & ACCURACY COMPARISON ---\n")
cm <- table(Actual = df$Gender, Predicted = df$Predicted)
print(cm)

acc <- sum(diag(cm)) / sum(cm)
cat(sprintf("\nDecision Tree & Rule Induction Model Accuracy: %.2f%%\n", acc * 100))
