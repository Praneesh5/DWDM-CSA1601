# EXP-36: Scatter Plot Analysis of Mobile Phones Sold vs Revenue

# Data Points (x: Mobile Phones Sold, y: Money in $)
mobile_data <- data.frame(
  PhonesSold = c(10, 15, 22, 28, 35, 42, 50, 60, 72, 85),
  MoneyEarned = c(5000, 7800, 11500, 14200, 18500, 22000, 26500, 31000, 37500, 44000)
)

cat("=====================================================\n")
cat("  EXP-36: SCATTER PLOT OF PHONES SOLD vs MONEY EARNED\n")
cat("=====================================================\n\n")

cat("Mobile Phone Sales Dataset (x, y points):\n")
print(mobile_data)
cat("\n")

correlation <- cor(mobile_data$PhonesSold, mobile_data$MoneyEarned)
cat(sprintf("Correlation between Phones Sold and Money Earned: %.4f\n\n", correlation))

# Generate Scatter Plot
png("mobile_sales_scatterplot_exp36.png", width=600, height=450)
plot(mobile_data$PhonesSold, mobile_data$MoneyEarned,
     col = "blue", pch = 19, cex = 1.5,
     main = "Scatter Plot: Mobile Phones Sold vs Money Earned",
     xlab = "Number of Mobile Phones Sold (x)",
     ylab = "Money Earned in $ (y)")
abline(lm(MoneyEarned ~ PhonesSold, data = mobile_data), col = "red", lwd = 2)
grid()
dev.off()
cat("Scatter plot saved as 'mobile_sales_scatterplot_exp36.png'\n")
