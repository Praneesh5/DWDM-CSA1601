# EXP-33: K-Means Cluster Size Comparison & Visualization

# Generate Clustering Dataset
set.seed(42)
x <- c(rnorm(25, mean = 20, sd = 3), rnorm(25, mean = 50, sd = 4), rnorm(25, mean = 80, sd = 3))
y <- c(rnorm(25, mean = 30, sd = 4), rnorm(25, mean = 75, sd = 3), rnorm(25, mean = 25, sd = 4))
cluster_data <- data.frame(X = x, Y = y)

cat("=====================================================\n")
cat("  EXP-33: K-MEANS CLUSTER ANALYSIS VARYING CLUSTER SIZE\n")
cat("=====================================================\n\n")

for (k in c(2, 3, 4, 5)) {
  km_model <- kmeans(cluster_data, centers = k, nstart = 10)
  cat(sprintf("--- K-Means with K = %d ---\n", k))
  cat(sprintf("Within-cluster sum of squares: %.2f\n", sum(km_model$withinss)))
  cat(sprintf("Total Between SS / Total SS:   %.2f%%\n\n", (km_model$betweenss / km_model$totss) * 100))
}

# Plotting Graph for K = 3
km_final <- kmeans(cluster_data, centers = 3, nstart = 10)
png("kmeans_variation_exp33.png", width=600, height=450)
plot(cluster_data$X, cluster_data$Y, col = km_final$cluster, pch = 19,
     main = "K-Means Clustering Visualization (K = 3)",
     xlab = "Feature X", ylab = "Feature Y")
points(km_final$centers, col = 1:3, pch = 4, cex = 3, lwd = 3)
dev.off()
cat("Cluster visualization plot saved as 'kmeans_variation_exp33.png'\n")
