# EXP-32: Customer Segmentation in Supermarket Mall using K-Means Clustering

# Dataset creation based on Mall Customers
df <- data.frame(
  CustomerID = 1:20,
  Age = c(19, 21, 20, 23, 31, 22, 35, 23, 64, 30, 67, 35, 58, 24, 37, 22, 35, 20, 52, 35),
  AnnualIncome = c(15, 15, 16, 16, 17, 17, 18, 18, 19, 19, 19, 19, 20, 20, 20, 20, 21, 21, 23, 23),
  SpendingScore = c(39, 81, 6, 77, 40, 76, 6, 94, 3, 72, 14, 99, 15, 77, 13, 79, 35, 66, 29, 98)
)

cat("=====================================================\n")
cat("  EXP-32: MALL CUSTOMER K-MEANS CLUSTERING (5 CLUSTERS)\n")
cat("=====================================================\n\n")

cat("Customer Dataset Preview:\n")
print(head(df, 10))
cat("\n")

# K-Means Clustering with K = 5 segments
set.seed(123)
km <- kmeans(df[, c("AnnualIncome", "SpendingScore")], centers = 5, nstart = 20)

df$Cluster <- factor(km$cluster)
cluster_labels <- c("Usual Customers", "Priority Customers", "Senior Citizen Target", "Young Target", "Balanced Target")
levels(df$Cluster) <- cluster_labels[1:length(levels(df$Cluster))]

cat("Cluster Centers (Annual Income vs Spending Score):\n")
print(km$centers)
cat("\n")

cat("Customer Cluster Segment Counts:\n")
print(table(df$Cluster))

# Plot Cluster Graph
png("mall_customer_clusters_exp32.png", width=650, height=450)
plot(df$AnnualIncome, df$SpendingScore, col = km$cluster, pch = 19, cex = 2,
     main = "Supermarket Customer Segmentation (5 Clusters)",
     xlab = "Annual Income (k$)", ylab = "Spending Score (1-100)")
points(km$centers, col = 1:5, pch = 8, cex = 3, lwd = 2)
legend("topright", legend = paste("Cluster", 1:5), col = 1:5, pch = 19)
dev.off()
cat("\nCluster Plot saved as 'mall_customer_clusters_exp32.png'\n")
