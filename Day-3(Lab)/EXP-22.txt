# EXP-22: Apriori and FP-Growth Algorithm

# Transactions Dataset from Image 1
# Customer 1: {a, d, e}, {a, b, c, e}
# Customer 2: {a, b, d, e}, {a, c, d, e}
# Customer 3: {b, c, e}, {b, d, e}
# Customer 4: {c, d}, {a, b, c}
# Customer 5: {a, d, e}, {a, b, e}

transactions <- list(
  T0001 = c("a", "d", "e"),
  T0024 = c("a", "b", "c", "e"),
  T0012 = c("a", "b", "d", "e"),
  T0031 = c("a", "c", "d", "e"),
  T0015 = c("b", "c", "e"),
  T0022 = c("b", "d", "e"),
  T0029 = c("c", "d"),
  T0040 = c("a", "b", "c"),
  T0033 = c("a", "d", "e"),
  T0038 = c("a", "b", "e")
)

N <- length(transactions)
min_sup <- 3        # support: 3 (30%)
min_conf <- 0.50    # confidence: 50%

cat("========================================
")
cat("  EXP-22: APRIORI AND FP-GROWTH ALGORITHM
")
cat("========================================
")
cat("Total Transactions:", N, "
")
cat("Minimum Support Count:", min_sup, "(30%)
")
cat("Minimum Confidence: 50%

")

# 1-Itemsets
items <- unlist(transactions)
freq1 <- table(items)
cat("--- 1-Itemsets Support Count (C1) ---
")
print(freq1)

L1 <- freq1[freq1 >= min_sup]
cat("
--- Frequent 1-Itemsets (L1) ---
")
print(L1)

get_support <- function(itemset, trans_list) {
  sum(sapply(trans_list, function(t) all(itemset %in% t)))
}

# 2-Itemsets
item_names <- names(L1)
pairs <- combn(item_names, 2, simplify = FALSE)
C2_sup <- sapply(pairs, get_support, trans_list = transactions)
names(C2_sup) <- sapply(pairs, function(p) paste(p, collapse = ", "))

cat("
--- 2-Itemsets Support Count (C2) ---
")
print(C2_sup)

L2_sup <- C2_sup[C2_sup >= min_sup]
cat("
--- Frequent 2-Itemsets (L2) ---
")
print(L2_sup)

# 3-Itemsets
triples <- combn(item_names, 3, simplify = FALSE)
C3_sup <- sapply(triples, get_support, trans_list = transactions)
names(C3_sup) <- sapply(triples, function(p) paste(p, collapse = ", "))

cat("
--- 3-Itemsets Support Count (C3) ---
")
print(C3_sup)

L3_sup <- C3_sup[C3_sup >= min_sup]
cat("
--- Frequent 3-Itemsets (L3) ---
")
print(L3_sup)

cat("
========================================
")
cat("  FP-GROWTH HEADER TABLE & ORDERED ITEMS
")
cat("========================================
")
ordered_items <- names(sort(L1, decreasing = TRUE))
cat("Ordered Frequent Items:", paste(ordered_items, collapse = " > "), "

")

cat("Ordered Transactions for FP-Tree:
")
for (id in names(transactions)) {
  t <- transactions[[id]]
  t_freq <- intersect(ordered_items, t)
  cat(id, ":", paste(t_freq, collapse = " -> "), "
")
}

cat("
========================================
")
cat("  ASSOCIATION RULES GENERATED (Conf >= 50%)
")
cat("========================================
")

all_frequent <- c(L2_sup, L3_sup)
rule_count <- 0

for (name in names(all_frequent)) {
  items_in_set <- unlist(strsplit(name, ", "))
  sup_AB <- all_frequent[name]
  
  len <- length(items_in_set)
  for (i in 1:(len - 1)) {
    lhs_combos <- combn(items_in_set, i, simplify = FALSE)
    for (lhs in lhs_combos) {
      rhs <- setdiff(items_in_set, lhs)
      sup_A <- get_support(lhs, transactions)
      conf <- sup_AB / sup_A
      if (conf >= min_conf) {
        rule_count <- rule_count + 1
        cat(sprintf("Rule %2d: {%s} => {%s} [Support = %d/10, Confidence = %.1f%%]
",
                    rule_count,
                    paste(lhs, collapse=", "),
                    paste(rhs, collapse=", "),
                    sup_AB, conf * 100))
      }
    }
  }
}
