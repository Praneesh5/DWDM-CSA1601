# EXP-26: WEKA Apriori & FP-Growth Algorithm (MONKEY/DONKEY Dataset)

# Database Transactions
transactions <- list(
  T1 = c("M", "O", "N", "K", "E", "Y"),
  T2 = c("D", "O", "N", "K", "E", "Y"),
  T3 = c("M", "A", "K", "E"),
  T4 = c("M", "U", "C", "K", "Y"),
  T5 = c("C", "O", "K", "I", "E")
)

N <- length(transactions)
min_sup <- 2  # 50% of 5
min_conf <- 0.80 # 80%

cat("=====================================================
")
cat("  EXP-26: WEKA APRIORI & FP-GROWTH ALGORITHM
")
cat("=====================================================
")
cat("Transactions: 5
")
cat("Minimum Support: 50% (Count >= 2)
")
cat("Minimum Confidence: 80%

")

cat("-----------------------------------------------------
")
cat("1. WEKA ARFF DATASET FORMAT REPRESENTATION
")
cat("-----------------------------------------------------
")
arff_content <- "@relation monkey_donkey_dataset

"
all_items <- sort(unique(unlist(transactions)))
for (item in all_items) {
  arff_content <- paste0(arff_content, "@attribute ", item, " {t, f}
")
}
arff_content <- paste0(arff_content, "
@data
")

for (id in names(transactions)) {
  t <- transactions[[id]]
  row_vals <- sapply(all_items, function(x) if (x %in% t) "t" else "f")
  arff_content <- paste0(arff_content, paste(row_vals, collapse=","), "
")
}
cat(arff_content, "
")

cat("-----------------------------------------------------
")
cat("2. APRIORI ALGORITHM EXECUTION
")
cat("-----------------------------------------------------
")

item_counts <- table(unlist(transactions))
cat("--- 1-Itemset Candidate Counts (C1) ---
")
print(item_counts)

L1 <- item_counts[item_counts >= min_sup]
cat("
--- Frequent 1-Itemsets (L1) ---
")
print(L1)

get_support <- function(itemset, trans_list) {
  sum(sapply(trans_list, function(t) all(itemset %in% t)))
}

freq_items <- names(L1)
pairs <- combn(freq_items, 2, simplify = FALSE)
C2_sup <- sapply(pairs, get_support, trans_list = transactions)
names(C2_sup) <- sapply(pairs, function(p) paste(p, collapse = ", "))

L2_sup <- C2_sup[C2_sup >= min_sup]
cat("
--- Frequent 2-Itemsets (L2) ---
")
print(L2_sup)

triples <- combn(freq_items, 3, simplify = FALSE)
C3_sup <- sapply(triples, get_support, trans_list = transactions)
names(C3_sup) <- sapply(triples, function(p) paste(p, collapse = ", "))

L3_sup <- C3_sup[C3_sup >= min_sup]
cat("
--- Frequent 3-Itemsets (L3) ---
")
print(L3_sup)

quads <- combn(freq_items, 4, simplify = FALSE)
C4_sup <- sapply(quads, get_support, trans_list = transactions)
names(C4_sup) <- sapply(quads, function(p) paste(p, collapse = ", "))

L4_sup <- C4_sup[C4_sup >= min_sup]
cat("
--- Frequent 4-Itemsets (L4) ---
")
print(L4_sup)

cat("
-----------------------------------------------------
")
cat("3. FP-GROWTH TREE CONSTRUCTION & ORDERED ITEMS
")
cat("-----------------------------------------------------
")
ordered_header <- names(sort(L1, decreasing = TRUE))
cat("FP-Growth Header Table (Frequency Order):", paste(ordered_header, collapse = " > "), "

")

cat("Transactions ordered by FP-Header frequency:
")
for (id in names(transactions)) {
  t <- transactions[[id]]
  t_ordered <- intersect(ordered_header, t)
  cat(sprintf("%s: %s
", id, paste(t_ordered, collapse = " -> ")))
}

cat("
-----------------------------------------------------
")
cat("4. GENERATED ASSOCIATION RULES (Conf >= 80%)
")
cat("-----------------------------------------------------
")
all_freq <- c(L2_sup, L3_sup, L4_sup)
rule_idx <- 0

for (name in names(all_freq)) {
  items_in_set <- unlist(strsplit(name, ", "))
  sup_AB <- all_freq[name]
  
  len <- length(items_in_set)
  for (i in 1:(len - 1)) {
    lhs_combos <- combn(items_in_set, i, simplify = FALSE)
    for (lhs in lhs_combos) {
      rhs <- setdiff(items_in_set, lhs)
      sup_A <- get_support(lhs, transactions)
      conf <- sup_AB / sup_A
      if (conf >= min_conf) {
        rule_idx <- rule_idx + 1
        cat(sprintf("Rule %2d: {%s} => {%s} [Supp: %d/5, Conf: %.0f%%]
",
                    rule_idx,
                    paste(lhs, collapse=","),
                    paste(rhs, collapse=","),
                    sup_AB, conf * 100))
      }
    }
  }
}
