# EXP-23: Market Basket Analysis, Apriori & FP-Growth Algorithm

# Market Basket Transactions Dataset from Image 2
transactions <- list(
  T1 = c("Milk", "Beer", "Diapers"),
  T2 = c("Bread", "Butter", "Milk"),
  T3 = c("Milk", "Diapers", "Cookies"),
  T4 = c("Bread", "Butter", "Cookies"),
  T5 = c("Beer", "Cookies", "Diapers"),
  T6 = c("Milk", "Diapers", "Bread", "Butter"),
  T7 = c("Bread", "Butter", "Diapers"),
  T8 = c("Beer", "Diapers"),
  T9 = c("Milk", "Diapers", "Bread", "Butter"),
  T10 = c("Beer", "Cookies")
)

all_items <- unique(unlist(transactions))
d <- length(all_items)

cat("=====================================================
")
cat("  EXP-23: MARKET BASKET ANALYSIS & QUESTION ANSWERS
")
cat("=====================================================
")
cat("Distinct Items (d =", d, "):", paste(all_items, collapse = ", "), "

")

# Question (a)
max_rules <- (3^d) - (2^(d + 1)) + 1
cat("-----------------------------------------------------
")
cat("(a) Maximum number of association rules (including zero support):
")
cat("    Formula: 3^d - 2^(d+1) + 1
")
cat("    Calculation: 3^", d, " - 2^", d+1, " + 1 = ", max_rules, " rules.
", sep="")
cat("-----------------------------------------------------

")

# Question (b)
max_trans_len <- max(sapply(transactions, length))
cat("-----------------------------------------------------
")
cat("(b) Maximum size of frequent itemsets (assuming minsup > 0):
")
cat("    Largest Transaction Size =", max_trans_len, "
")
cat("    Maximum size of frequent itemsets = ", max_trans_len, "
", sep="")
cat("-----------------------------------------------------

")

cat("=====================================================
")
cat("  APRIORI & FP-GROWTH ALGORITHM (minsup = 3, minconf = 50%)
")
cat("=====================================================

")

N <- length(transactions)
min_sup <- 3
min_conf <- 0.50

freq1 <- table(unlist(transactions))
cat("--- 1-Itemsets Support Count (C1) ---
")
print(freq1)

L1 <- freq1[freq1 >= min_sup]
cat("
--- Frequent 1-Itemsets (L1) ---
")
print(L1)

get_support <- function(itemset, trans_list) {
  sum(sapply(trans_list, function(t) all(itemset %in% t)))
}

item_names <- names(L1)
pairs <- combn(item_names, 2, simplify = FALSE)
C2_sup <- sapply(pairs, get_support, trans_list = transactions)
names(C2_sup) <- sapply(pairs, function(p) paste(p, collapse = ", "))

cat("
--- 2-Itemsets Support Count (C2) ---
")
print(C2_sup)

L2_sup <- C2_sup[C2_sup >= min_sup]
cat("
--- Frequent 2-Itemsets (L2) ---
")
print(L2_sup)

triples <- combn(item_names, 3, simplify = FALSE)
C3_sup <- sapply(triples, get_support, trans_list = transactions)
names(C3_sup) <- sapply(triples, function(p) paste(p, collapse = ", "))

L3_sup <- C3_sup[C3_sup >= min_sup]
cat("
--- Frequent 3-Itemsets (L3) ---
")
print(L3_sup)

ordered_items <- names(sort(L1, decreasing = TRUE))
cat("
--- FP-Growth Header Table Order ---
")
cat("Items ordered by frequency:", paste(ordered_items, collapse = " > "), "
")

cat("
--- Association Rules (Confidence >= 50%) ---
")
all_frequent <- c(L2_sup, L3_sup)
rule_count <- 0

for (name in names(all_frequent)) {
  items_in_set <- unlist(strsplit(name, ", "))
  sup_AB <- all_frequent[name]
  
  len <- length(items_in_set)
  for (i in 1:(len - 1)) {
    lhs_combos <- combn(items_in_set, i, simplify = FALSE)
    for (lhs in lhs_combos) {
      rhs <- setdiff(items_in_set, lhs)
      sup_A <- get_support(lhs, transactions)
      conf <- sup_AB / sup_A
      if (conf >= min_conf) {
        rule_count <- rule_count + 1
        cat(sprintf("Rule %2d: {%s} => {%s} [Support = %d/10, Confidence = %.1f%%]
",
                    rule_count,
                    paste(lhs, collapse=", "),
                    paste(rhs, collapse=", "),
                    sup_AB, conf * 100))
      }
    }
  }
}
