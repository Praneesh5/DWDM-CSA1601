# EXP-31: Association Rule Analysis, Apriori & FP-Growth Algorithm

transactions <- list(
  T1 = c("Milk", "Beer", "Diapers"),
  T2 = c("Bread", "Butter", "Milk"),
  T3 = c("Milk", "Diapers", "Cookies"),
  T4 = c("Bread", "Butter", "Cookies"),
  T5 = c("Beer", "Cookies", "Diapers"),
  T6 = c("Milk", "Diapers", "Bread", "Butter"),
  T7 = c("Bread", "Butter", "Diapers"),
  T8 = c("Beer", "Diapers"),
  T9 = c("Milk", "Diapers", "Bread", "Butter"),
  T10 = c("Beer", "Cookies")
)

all_items <- unique(unlist(transactions))
d <- length(all_items)

cat("=====================================================\n")
cat("  EXP-31: MARKET BASKET ANALYSIS & QUESTION ANSWERS\n")
cat("=====================================================\n")
cat("Distinct Items (d =", d, "):", paste(all_items, collapse = ", "), "\n\n")

# Question (a)
max_rules <- (3^d) - (2^(d + 1)) + 1
cat("-----------------------------------------------------\n")
cat("(a) Maximum number of association rules (including zero support):\n")
cat("    Formula: 3^d - 2^(d+1) + 1\n")
cat("    Calculation: 3^", d, " - 2^", d+1, " + 1 = ", max_rules, " rules.\n", sep="")
cat("-----------------------------------------------------\n\n")

# Question (b)
max_trans_len <- max(sapply(transactions, length))
cat("-----------------------------------------------------\n")
cat("(b) Maximum size of frequent itemsets (assuming minsup > 0):\n")
cat("    Largest Transaction Size =", max_trans_len, "\n")
cat("    Maximum size of frequent itemsets = ", max_trans_len, "\n", sep="")
cat("-----------------------------------------------------\n\n")

freq1 <- table(unlist(transactions))
L1 <- freq1[freq1 >= 3]
cat("Frequent 1-Itemsets (L1):\n")
print(L1)

get_support <- function(itemset, trans_list) {
  sum(sapply(trans_list, function(t) all(itemset %in% t)))
}

item_names <- names(L1)
pairs <- combn(item_names, 2, simplify = FALSE)
C2_sup <- sapply(pairs, get_support, trans_list = transactions)
names(C2_sup) <- sapply(pairs, function(p) paste(p, collapse = ", "))
L2_sup <- C2_sup[C2_sup >= 3]

cat("\nFrequent 2-Itemsets (L2):\n")
print(L2_sup)

triples <- combn(item_names, 3, simplify = FALSE)
C3_sup <- sapply(triples, get_support, trans_list = transactions)
names(C3_sup) <- sapply(triples, function(p) paste(p, collapse = ", "))
L3_sup <- C3_sup[C3_sup >= 3]

cat("\nFrequent 3-Itemsets (L3):\n")
print(L3_sup)
