# EXP-28: ARFF Dataset Creation, Apriori & FP-Growth Rule Evaluation

# Dataset from Image 4
transactions <- list(
  T1 = c("Hot_Dogs", "Buns", "Ketchup"),
  T2 = c("Hot_Dogs", "Buns"),
  T3 = c("Hot_Dogs", "Coke", "Chips"),
  T4 = c("Chips", "Coke"),
  T5 = c("Chips", "Ketchup"),
  T6 = c("Hot_Dogs", "Coke", "Chips")
)

N <- length(transactions)
min_sup <- 2        # 33.33% of 6
min_conf <- 0.60    # 60%

cat("=====================================================
")
cat("  EXP-28: ARFF DATASET & RULE EVALUATION (ACCEPTED / REJECTED)
")
cat("=====================================================

")

cat("-----------------------------------------------------
")
cat("1. ARFF FILE FORMAT DEFINITION
")
cat("-----------------------------------------------------
")
all_items <- c("Hot_Dogs", "Buns", "Ketchup", "Coke", "Chips")
arff_str <- "@relation grocery_market_basket

"
for (item in all_items) {
  arff_str <- paste0(arff_str, "@attribute ", item, " {t, f}
")
}
arff_str <- paste0(arff_str, "
@data
")

for (id in names(transactions)) {
  t <- transactions[[id]]
  vals <- sapply(all_items, function(x) if (x %in% t) "t" else "f")
  arff_str <- paste0(arff_str, paste(vals, collapse=","), "
")
}
cat(arff_str, "
")

cat("-----------------------------------------------------
")
cat("2. FREQUENT ITEMSETS (minsup = 33.33% = 2/6)
")
cat("-----------------------------------------------------
")

freq1 <- table(unlist(transactions))
L1 <- freq1[freq1 >= min_sup]
cat("Frequent 1-Itemsets (L1):
")
print(L1)

get_support <- function(itemset, trans_list) {
  sum(sapply(trans_list, function(t) all(itemset %in% t)))
}

freq_items <- names(L1)
pairs <- combn(freq_items, 2, simplify = FALSE)
C2_sup <- sapply(pairs, get_support, trans_list = transactions)
names(C2_sup) <- sapply(pairs, function(p) paste(p, collapse = ", "))
L2_sup <- C2_sup[C2_sup >= min_sup]

cat("
Frequent 2-Itemsets (L2):
")
print(L2_sup)

triples <- combn(freq_items, 3, simplify = FALSE)
C3_sup <- sapply(triples, get_support, trans_list = transactions)
names(C3_sup) <- sapply(triples, function(p) paste(p, collapse = ", "))
L3_sup <- C3_sup[C3_sup >= min_sup]

cat("
Frequent 3-Itemsets (L3):
")
print(L3_sup)

cat("
-----------------------------------------------------
")
cat("3. FP-GROWTH TREE ORDERING
")
cat("-----------------------------------------------------
")
ordered_fp <- names(sort(L1, decreasing = TRUE))
cat("Item Order:", paste(ordered_fp, collapse = " > "), "
")

cat("
-----------------------------------------------------
")
cat("4. ASSOCIATION RULES & ACCEPTED/REJECTED CLASSIFICATION
")
cat("-----------------------------------------------------
")
cat("Rule Thresholds: Minimum Support = 33.33% (2/6), Minimum Confidence = 60.00%

")

all_freq <- c(L2_sup, L3_sup)
rule_id <- 0

for (name in names(all_freq)) {
  items_in_set <- unlist(strsplit(name, ", "))
  sup_AB <- all_freq[name]
  
  len <- length(items_in_set)
  for (i in 1:(len - 1)) {
    lhs_combos <- combn(items_in_set, i, simplify = FALSE)
    for (lhs in lhs_combos) {
      rhs <- setdiff(items_in_set, lhs)
      sup_A <- get_support(lhs, transactions)
      conf <- sup_AB / sup_A
      status <- if (conf >= min_conf) "ACCEPTED" else "REJECTED"
      
      rule_id <- rule_id + 1
      cat(sprintf("Rule %2d: {%s} => {%s} | Supp: %d/6 (%.1f%%) | Conf: %.1f%% | Status: [%s]
",
                  rule_id,
                  paste(lhs, collapse=","),
                  paste(rhs, collapse=","),
                  sup_AB, (sup_AB/N)*100,
                  conf * 100, status))
    }
  }
}
