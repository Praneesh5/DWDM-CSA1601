# EXP-25: Linear Regression and Multiple Regression on Diabetes Dataset

# Load or generate sample diabetes dataset
csv_path <- "C:/Users/prane/Downloads/diabetes.csv"

if (file.exists(csv_path)) {
  diabetes <- read.csv(csv_path)
} else {
  set.seed(123)
  n <- 150
  age <- sample(21:75, n, replace = TRUE)
  glucose <- round(70 + 0.8 * age + rnorm(n, mean = 20, sd = 15))
  bp <- round(60 + 0.3 * age + rnorm(n, mean = 10, sd = 8))
  bmi <- round(20 + 0.15 * age + rnorm(n, mean = 5, sd = 4), 1)
  insulin <- round(50 + 1.2 * age + rnorm(n, mean = 30, sd = 20))
  outcome <- ifelse(glucose > 120 | bmi > 30, 1, 0)
  
  diabetes <- data.frame(Age = age, Glucose = glucose, BloodPressure = bp, BMI = bmi, Insulin = insulin, Outcome = outcome)
}

cat("=====================================================
")
cat("  EXP-25: DIABETES TREND ANALYSIS & REGRESSION
")
cat("=====================================================
")
cat("Dataset Preview (First 6 rows):
")
print(head(diabetes))
cat("
Total Records:", nrow(diabetes), "

")

cat("-----------------------------------------------------
")
cat("1. DIABETES TREND ANALYSIS BY AGE GROUP
")
cat("-----------------------------------------------------
")
diabetes$AgeGroup <- cut(diabetes$Age,
                         breaks = c(20, 30, 40, 50, 60, 80),
                         labels = c("20-30", "31-40", "41-50", "51-60", "61+"))

age_summary <- aggregate(cbind(Glucose, BloodPressure, BMI) ~ AgeGroup, data = diabetes, mean)
cat("Average Health Metrics by Age Group:
")
print(age_summary)
cat("
")

cat("-----------------------------------------------------
")
cat("2. SIMPLE LINEAR REGRESSION (Glucose ~ Age)
")
cat("-----------------------------------------------------
")
simple_model <- lm(Glucose ~ Age, data = diabetes)
summary_simple <- summary(simple_model)
print(summary_simple)

cat(sprintf("
Simple Linear Regression Equation:
"))
cat(sprintf("Glucose = %.4f + (%.4f * Age)
", coef(simple_model)[1], coef(simple_model)[2]))
cat(sprintf("R-squared: %.4f, p-value: %.4e

", summary_simple$r.squared, summary_simple$coefficients[2, 4]))

cat("-----------------------------------------------------
")
cat("3. MULTIPLE LINEAR REGRESSION (Glucose ~ Age + BMI + BloodPressure + Insulin)
")
cat("-----------------------------------------------------
")
multi_model <- lm(Glucose ~ Age + BMI + BloodPressure + Insulin, data = diabetes)
summary_multi <- summary(multi_model)
print(summary_multi)

cat(sprintf("
Multiple Linear Regression Equation:
"))
cat(sprintf("Glucose = %.4f + (%.4f * Age) + (%.4f * BMI) + (%.4f * BloodPressure) + (%.4f * Insulin)
",
            coef(multi_model)[1], coef(multi_model)[2], coef(multi_model)[3], coef(multi_model)[4], coef(multi_model)[5]))
cat(sprintf("Multiple R-squared: %.4f, Adjusted R-squared: %.4f

", summary_multi$r.squared, summary_multi$adj.r.squared))

cat("-----------------------------------------------------
")
cat("4. PREDICTION FOR A 45-YEAR-OLD INDIVIDUAL
")
cat("-----------------------------------------------------
")
new_patient <- data.frame(Age = 45, BMI = 28.5, BloodPressure = 75, Insulin = 120)
pred_simple <- predict(simple_model, newdata = new_patient)
pred_multi  <- predict(multi_model, newdata = new_patient)

cat(sprintf("Predicted Glucose Level (Simple Model):   %.2f mg/dL
", pred_simple))
cat(sprintf("Predicted Glucose Level (Multiple Model): %.2f mg/dL
", pred_multi))
